    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, createUserWithEmailAndPassword, onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, collection, getDocs, query, orderBy } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // YOUR REAL FIREBASE CONFIGURATION
        const firebaseConfig = {
            apiKey: "AIzaSyAQnY1nrC_j8PS_qs4_rOozaUFN10Re0JI",
            authDomain: "conflict-of-interest-389af.firebaseapp.com",
            projectId: "conflict-of-interest-389af",
            storageBucket: "conflict-of-interest-389af.firebasestorage.app",
            messagingSenderId: "940388335483",
            appId: "1:940388335483:web:17c6e489cc52298d9847de",
            measurementId: "G-LQRXQ2M0BE"
        };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        // --- Fetch Articles from Database ---
        const articlesGrid = document.getElementById('articles-grid');
        const loadingText = document.getElementById('loading-text');
        let allArticles = [];

        async function loadArticles() {
            try {
                const q = query(collection(db, "articles"), orderBy("createdAt", "desc"));
                const querySnapshot = await getDocs(q);
                
                loadingText.style.display = 'none';
                
                querySnapshot.forEach((doc) => {
                    const data = doc.data();
                    allArticles.push({ id: doc.id, ...data });
                    
                    // Create HTML for each article
                    const articleHTML = `
                        <article class="flex flex-col group cursor-pointer article-item" data-category="${data.category}">
                            <div class="overflow-hidden rounded-sm mb-6 border border-brand-border bg-brand-card h-64 relative">
                                <img src="${data.imageUrl}" alt="${data.category}" class="w-full h-full object-cover hover-zoom-img opacity-80 group-hover:opacity-100 transition-opacity">
                            </div>
                            <div class="flex items-center space-x-3 mb-4">
                                <span class="text-xs font-bold uppercase tracking-widest text-brand-accent">${data.category}</span>
                                <span class="text-brand-border">|</span>
                                <span class="text-xs font-bold text-white uppercase bg-brand-border/30 px-2 py-1 rounded-sm">${data.readTime || '5 Min'}</span>
                            </div>
                            <div class="flex-grow">
                                <h3 class="text-2xl font-serif font-bold leading-snug mb-4 text-white group-hover:text-brand-accent transition-colors">${data.title}</h3>
                            </div>
                        </article>
                    `;
                    articlesGrid.insertAdjacentHTML('beforeend', articleHTML);
                });

                attachArticleClickHandlers();
            } catch (error) {
                loadingText.textContent = "Welcome to Conflict of Interest. (Publish an article in your admin panel to see it here!)";
                console.log("Firestore empty or error:", error);
            }
        }
        
        loadArticles();

        // --- View Routing ---
        const homeView = document.getElementById('home-view');
        const articleView = document.getElementById('article-view');
        const aboutView = document.getElementById('about-view');

        function showView(viewToShow) {
            [homeView, articleView, aboutView].forEach(v => v.classList.add('hidden'));
            viewToShow.classList.remove('hidden');
            window.scrollTo(0, 0);
        }

        document.getElementById('nav-about-link').addEventListener('click', (e) => { e.preventDefault(); showView(aboutView); });
        document.getElementById('nav-home-link').addEventListener('click', (e) => { e.preventDefault(); showView(homeView); });
        document.getElementById('nav-logo').addEventListener('click', (e) => { e.preventDefault(); showView(homeView); });
        
        document.body.addEventListener('click', (e) => {
            if (e.target.closest('.nav-home-trigger')) {
                e.preventDefault();
                showView(homeView);
            }
        });

        function attachArticleClickHandlers() {
            const articleItems = document.querySelectorAll('.article-item');
            articleItems.forEach((item, index) => {
                item.addEventListener('click', () => {
                    const articleData = allArticles[index];
                    document.getElementById('view-title').textContent = articleData.title;
                    document.getElementById('view-category').textContent = articleData.category;
                    document.getElementById('view-author').textContent = `By ${articleData.author}`;
                    
                    // Format content (apply drop cap class to first paragraph)
                    let formattedContent = `<p class="first-letter:text-7xl first-letter:font-bold first-letter:text-white first-letter:mr-3 first-letter:float-left first-letter:leading-[0.8] mb-6">${articleData.content.replace(/\n\n/g, '</p><p class="mb-6">')}</p>`;
                    document.getElementById('view-content').innerHTML = formattedContent;
                    
                    showView(articleView);
                });
            });
        }

        // --- Filtering Logic ---
        const dropdownBtn = document.getElementById('dropdown-button');
        const dropdownMenu = document.getElementById('dropdown-menu');
        
        dropdownBtn.addEventListener('click', () => dropdownMenu.classList.toggle('hidden'));
        
        document.querySelectorAll('.filter-btn').forEach(btn => {
            btn.addEventListener('click', (e) => {
                e.preventDefault();
                showView(homeView);
                const filter = btn.getAttribute('data-filter');
                dropdownMenu.classList.add('hidden');
                
                document.querySelectorAll('.article-item').forEach(article => {
                    const cats = article.getAttribute('data-category');
                    if (filter === 'All' || (cats && cats.includes(filter))) {
                        article.style.display = '';
                    } else {
                        article.style.display = 'none';
                    }
                });
            });
        });

        // --- Auth & Subscription ---
        const subForm = document.getElementById('subscribe-form');
        const subMsg = document.getElementById('sub-msg');
        const headerAuthBtn = document.getElementById('header-auth-btn');

        subForm.addEventListener('submit', async (e) => {
            e.preventDefault();
            const email = document.getElementById('sub-email').value;
            const pass = document.getElementById('sub-pass').value;
            document.getElementById('sub-btn').textContent = "Processing...";
            
            try {
                await createUserWithEmailAndPassword(auth, email, pass);
                subForm.style.display = 'none';
                subMsg.textContent = "ACCESS GRANTED. Welcome to the ledger.";
                subMsg.classList.remove('hidden');
            } catch (err) {
                subMsg.textContent = "Error: " + err.message;
                subMsg.classList.remove('hidden');
                document.getElementById('sub-btn').textContent = "Sign Up";
            }
        });

        onAuthStateChanged(auth, (user) => {
            if (user) {
                headerAuthBtn.textContent = "Sign Out";
                headerAuthBtn.href = "#";
                headerAuthBtn.addEventListener('click', () => signOut(auth));
                subForm.style.display = 'none';
                subMsg.textContent = "You are securely authenticated.";
                subMsg.classList.remove('hidden');
            } else {
                headerAuthBtn.textContent = "Subscribe";
                headerAuthBtn.href = "#subscribe";
                subForm.style.display = 'flex';
                subMsg.classList.add('hidden');
            }
        });
    </script>
